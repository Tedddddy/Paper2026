\begin{definition}
    Let $T = (S,R,V)$ be a transition system, where $S$ is the set of worlds, 
    $R \subseteq S \times S$ and $V$ a valuation. For each agent $i$, we define a probability transitions 
    $P_i(s,s')$ such that $P_i(s,s') > 0 \implies sRs'$ and $\sum_{s'}^{} P_i(s,s') = 1$. \newline
    The set of paths generated by the underyling transistion systems is also called Paths($T$).

\end{definition}
\begin{definition}
    Let $\sigma = s0s1...sn$ be a finite prefix. Let $C(\sigma) = \{\pi \in \mbox{ Paths(T)} \mid \sigma \mbox{ is prefix of } \pi\}$.
    Then for each finite prefix and agent  $i$ we define the initial distribution of agent $i$ to be $\mu_i^0(C(\sigma)) = \omega(s0) \cdot P(s0,s1) \cdot
    ... P_i(s_{n-1},s_n)$.

\end{definition}

\begin{definition}
   Let $\pi$ be a path and $t$ a time. We define $C_i(\pi, t) = \{   \pi' = s_0's_1' ... \in Paths(T) \mid
    s_k \sim_i s_k' \mbox{ for all } 0 \leq k \leq t \}$. Furthermore, given 
    a set $A \subseteq Paths(T)$ the subjective posterior is \[
    \mu_i^{\pi,t}(A) = \frac{\mu_i^0 (A \cap C_i (\pi,t))} {\mu_i^0(C_i(\pi,t))}     \]
\end{definition}

\begin{definition}
    Let $T$ be a transition system, $\pi \in \mathrm{Paths}(T)$ and $t$ a timepoint. Then we have:
    \[
        \begin{aligned}
            T, \pi, t &\vDash p
                &&\text{iff } \pi(t) \in V(p), \\
            T, \pi, t &\vDash B_i^{[l,u]}\phi
                &&\text{iff } \mu_i^{\pi,t}(A_t(\phi)) \in [l,u] \mbox{ where } A_t(\phi) = \{\pi' \in Paths(T) \mid T, \pi', t \vDash \phi\}
        \end{aligned}
    \]
\end{definition}
