\begin{definition}
    Let $Var$ be the set of propositional variables. Then $\phi$ is defined inductively as follows: 
    \[
       \phi ::=  p \mid \phi \lor \phi \mid X\phi \mid  \phi U \psi \mid \langle a \rangle \phi 
    \]
\end{definition}

\begin{definition}
    Let  $\phi $ be a formula and $p \in Var$. Then the Closure of $\phi$ is inductively defined as

\[
    \begin{aligned}
        Cl(p) &= Cl(\neg p) = \{p,\neg p\} \\
        Cl(\neg \phi) &= Cl(\phi) \\
        Cl(\phi \lor \psi) &= \{\phi \lor \psi, \neg(\phi \lor \psi)\} \cup  Cl(\phi) \cup Cl(\psi) \\
        Cl(X \phi) &= \{X \phi, \neg X \phi\} \cup Cl(\phi) \\
        Cl(\phi U \psi) &= \{\phi U \psi, \neg(\phi U \psi), true U  \langle a \rangle \psi, \neg (true U \langle a \rangle \psi)\} \cup Cl(\phi) \cup Cl(\psi) \\
        Cl(\langle a \rangle \phi) &= \{\langle a \rangle \phi, \neg \langle a \rangle \phi\} \cup Cl(\phi)
    \end{aligned}
\]
\end{definition}



\begin{definition}
An atom $A$ is a subset of $Cl(\phi)$ such that it is consistent that means $A \nvdash \bot$ and 
\[
\begin{aligned}
&\text{if } \psi \in Cl(\phi),
&&\text{ then } \psi \in A \text{ iff } \neg \psi \notin A, \\
&\text{if } \phi_1 \lor \phi_2 \in Cl(\phi), 
&&\text{ then } \phi_1 \lor \phi_2 \in A\text{ iff } \phi_1 \in A \text{ or } \phi_2 \in A \\
&\text{if }  \phi_1 U \phi_2 \in Cl(\phi),
&&\text{then  if } \phi_1 U \phi_2 \in A \mbox{ then } \psi_1 \in A \mbox{ or } \psi_2 \in A \\   
&\text{if }  \phi_1 U \phi_2 \in Cl(\phi),
&&\text{then  if } \phi_2 \in A \mbox{ then } \phi_1 U \phi_2 \in A
\end{aligned}
\]
\end{definition}

\begin{lemma}
    Assume $A_i \vdash \psi$ For some $\psi \in Cl(\phi)$. Then $\psi \in A_i$.

    \begin{proof}
        Assume $\psi \notin A_i$. That means $\psi \in A_i$. But then $A_i \vdash \psi$ and $A_i \vdash \neg \psi$ which is a contradiction.
    \end{proof}
\end{lemma}

\begin{definition}
    We say two atoms $A,B$ are in relation $R_x (A R_x B)$ iff

\[
\begin{aligned}
&\text{if } X\psi \in Cl(\phi),
&&\text{ then } X\psi \in A \text{ iff } \psi \in B, \\[2mm]
&\text{if } \psi_1 U \psi_2 \in Cl(\phi),
&&\text{ then if } \psi_1 U \psi_2, \neg\psi_2 \in A,
   \text{ then } \psi_1 U \psi_2 \in B, \\
&&&\text{ and if } \psi_1 U \psi_2 \in B, \psi_1 \in A,
   \text{ then } \psi_1 U \psi_2 \in A.
\end{aligned}
\]
\end{definition}

\begin{lemma}

For any atom $A$ there exists an atom $B$ s.t. $A R_x B$.

\begin{proof}
    Assume $A$ is an atom. Then we define $B'$ as follows: 
    \[
        B' = \{ \phi \mid X \phi \in A\} \cup \{ \neg \phi \mid \neg X \phi \in A\} \cup \{\psi_1 U \psi_2 \mid \psi_1 U \psi_2, \neg \psi_2 \in A\}
        \cup \{   \neg(\psi_1 U \psi_2) \mid \neg(\psi_1 U \psi_2), \psi_1 \in A  \}
    \] 

    Now we show that $B$ is consistent. Assume it is not. Then there exists $\alpha_1,..., \alpha_n$ s.t.
    $B \vdash \alpha_1 \land \alpha_2 ... \land \alpha_n \rightarrow \bot$. 
    Let's define $\hat{B} = \bigwedge B$. It follows:
    \[
        \begin{array}{rll} 
            & \vdash \hat{B} \land \alpha_1 \land ... \land\alpha_n  \rightarrow \bot \\
            & \vdash X(\hat{B} \land \alpha_1 \land ... \land\alpha_n  \rightarrow \bot )
            & \text{By X-necessitation} \\
            & \vdash X(\hat{B} \land \alpha_1 \land ... \land\alpha_n ) \rightarrow X \bot
            & \text{F2} \\
            & \vdash X(\hat{B}) \land X\alpha_1 \land ... \land X\alpha_n \rightarrow X \bot \\
        \end{array}
    \]

    Assume, $a_i = \phi$. Then by definition we have $A \vdash Xa_i$. If $a_i = \neg \phi$, then
    we can use F1 to obtain $A \vdash X \neg \phi$. If $a_i = \psi_1 U \psi_2$ then 
    by definition, $\psi_1 U \psi_2 \in A$ and $\neg \psi_2 \in A$. Consider the following proof steps:

       \[
        \begin{array}{rll} 
            & A \vdash \neg \psi_2 \\
            & A \vdash \psi_1 U \psi_2 \\
            & A \vdash \psi_1 U \psi_2 \land \neg \psi_2 \\
            & A \vdash \psi_1 U \psi_2 \land \neg \psi_2 \rightarrow (\psi_2 \lor (\psi_1 \land X ( \psi_1 U \psi_2 ))) \land \neg \psi_2 
            & \text{Unfolding U} \\
            & A \vdash  \psi_1 U \psi_2 \land \neg \psi_2 \rightarrow (\psi_1 \land X(\psi_1 U \psi_2))
            & \text{distributivity and other rules} \\
            & A \vdash \psi_1 \land X(\psi_1 U \psi_2) \\
            & A \vdash X(\psi_1 U \psi_2).

        \end{array}
        \]

        Hence, $A \vdash Xa_i$. If $a_i = \neg (\psi_1 U \psi_2)$, then by
        definition, $\neg(\psi_1 U \psi_2), \psi_1 \in A.$ Our goal is to derive
        $A \vdash \neg(\psi_1 U \psi_2) \land \psi_1 \rightarrow X \neg (\psi_1 U \psi_2)$.

        \[
        \begin{array}{rll} 
                & A \vdash \neg(\psi_1 U \psi_2) \land \psi_1 \\
                & A \vdash \neg(\psi_2 \lor (\psi_1 \land X(\psi_1 U \psi_2))) \land \psi_1
                & \text{Unfolding U} \\
                & A \vdash \neg \psi_2 \land (\neg \psi_1 \lor \neg X(\psi_1 U \psi_2)) \land \psi_1 \\
                & A \vdash \neg \psi_1 \lor \neg X(\psi_1 U \psi_2) \\
                & \neg X (\psi_1 U \psi_2) 
                & \text{$\psi_1$ is derivable} \\
                & A \vdash X \neg (\psi_1 U \psi_2) 
                & \text{F1}

        \end{array}
        \]

        
        By similar argument, we can conclude $A \vdash X\hat{B}$. So it follows 
        \[
            A \vdash X\bot
        \] 
        Because $X \bot \rightarrow \bot$ is a tautology, we conclude $A \vdash \bot$ which is a contradiction. 
        Now, by the Extension Lemma, we can extend $B'$ to an atom $B$. It is easy to check, that $A R_x B$.
        

\end{proof}

\end{lemma}

\end{cases}
\]

\begin{lemma}
    Let $C,D$ be some atoms such that $(C, D) \notin R_x$. Then
    \[
        \vdash \hat{C} \rightarrow X \neg \hat{D}
    \]

    \begin{proof}
        There are different reasons for $(C, D) \notin R_x$. First, there is a formula $X \psi \in Cl(\phi)$ s.t.
        $X \psi \in C$ and $\psi \notin D$, i.e., $\neg \psi \in D$.

        \[
            \begin{array}{rll} 
                    &  \vdash \hat{C} \rightarrow X \psi \\
                    &  \vdash  \psi \rightarrow \neg \neg \psi \\
                    &  \vdash X \psi \rightarrow X \, \neg \neg \psi 
                    &  \text{X is monotonic + 2 } \\
                    &  \vdash \hat{C} \rightarrow X \neg \neg \psi \\
                    &  \vdash \hat{D} \rightarrow \neg \psi \\
                    &  \vdash \neg \neg \psi \rightarrow \neg \hat{D} \\
                    &  \vdash X \, \neg \neg \psi \rightarrow X \neg \hat{D} \\
                    &  \hat{C} \rightarrow X \neg \hat{D}



            \end{array}
        \]

        Second case is $X \psi \notin A$ i.e., $\neg X \in C$ but $\psi \in D$.

                \[
            \begin{array}{rll} 
                    &  \vdash \hat{C} \rightarrow \neg X \psi \\
                    &  \vdash  X \psi \leftrightarrow \neg X \psi \\
                    &  \vdash \neg X \psi \rightarrow X \, \neg \psi \\
                    &  \vdash \hat{C} \rightarrow X \neg \psi \\
                    &  \vdash \hat{D} \rightarrow \psi \\
                    &  \vdash \neg \psi \rightarrow \neg \hat{D } \\
                    &  \vdash X \,\neg \psi \rightarrow X \neg \hat{D} \\
                    &  \hat{C} \rightarrow X \neg \hat{D}


            \end{array}
        \]

        Third, there is a formula $\psi_1 U \psi_2$ s.t. $\psi_1 U \psi_2, \neg \psi_2 \in A$ and $\neg(\psi_1 U \psi_2) \in B$

                        \[
            \begin{array}{rll} 
                    &  \vdash \hat{C} \rightarrow (\psi_1 U \psi_2 \land \neg \psi_2) \\
                    &  \vdash  \psi_1 U \psi_2 \leftrightarrow \psi_2 \lor (\psi_1 \land X(\psi_1 U \psi_2)) \\
                    &  \vdash \hat{C} \rightarrow X (\psi_1 U \psi_2) \\
                    &  \vdash \hat{D} \rightarrow \neg (\psi_1 U \psi_2) \\
                    &  \vdash \neg \neg (\psi_1 U \psi_2) \rightarrow \neg \hat{B} \\
                    &  \vdash \psi_1 U \psi_2 \rightarrow \neg \neg (\psi_1 U \psi_2) \\
                    &  \vdash \psi_1 U \psi_2 \rightarrow X \neg \hat{D} \\
                    &  X (\psi_1 U \psi_2) \rightarrow X \neg \hat{D} \\
                    &  \hat{C} \rightarrow X \neg \hat{D}


            \end{array}
        \]

        Last, we have $\psi_1 U \psi_2 \in D$, $\psi_1 \in C$ and $\psi_1 U \psi_2 \notin C$, i.e., $\neg(\psi_1 U \psi_2) \in C$.
        From unraveling $\vdash \neg(\psi_1 U \psi_2) \rightarrow (\neg \psi_1 \lor \neg \, X(\psi_1 U \psi_2))$. Since $\vdash \hat{C} \rightarrow \neg X(\psi_1 U \psi_2)$
    \end{proof}
\end{lemma}

\begin{lemma}
    Let $C$ be some atom. Then
    \[  
        \vdash \hat{C} \rightarrow X (\bigvee_{D \in \mathcal{C} \text{ and } C R_x D} \hat{D})
    \]

    \begin{proof}
        We define $\mathcal{C} = S \cup \bar{S}$ where $S = \{D \in \mathcal{C} \mid (C,D) \in R_x \}$ and $\bar{S} = \{D \in \mathcal{C} \mid  (C,D) \notin R_x\}$.
        By Lemma 11 we have 
        \[
            \begin{array}{rll}
                & \vdash \hat{C} \rightarrow \, \bigcirc (\bigvee_{D \in S} \hat{D} \lor \bigvee_{D \in \bar{S}} \hat{D})
                &\text{Lemma 11} \\
                & \vdash \hat{C} \rightarrow \bigwedge_{D \in \bar{S}} \bigcirc \, \neg \hat{D} 
                &\text{Lemma 12} \\
                & \vdash \hat{C} \rightarrow \bigcirc (\bigwedge_{D \in \bar{S}} \, \neg \hat{D}) 
                & \text{$\bigcirc (\phi \land \psi) \leftrightarrow \bigcirc \phi \land \bigcirc \psi$} \\
                & \vdash \hat{C} \rightarrow  \, \bigcirc (\bigvee_{D \in S} \hat{D} \lor \bigvee_{D \in \bar{S}} \hat{D}) \land
                \bigcirc (\bigwedge_{D \in \bar{S}} \, \neg \hat{D}) \\
                & \vdash \rightarrow \hat{C} \bigcirc  (\bigvee_{D \in S} \hat{D} \lor \bigvee_{D \in \bar{S}} \hat{D}) \land
                (\bigwedge_{D \in \bar{S}} \, \neg \hat{D}))
            \end{array}
        \]


    \end{proof}
 \end{lemma}

\begin{lemma}

    \begin{proof}
        
        Assume $\langle b \rangle \psi \in G_i$.
        \[
                \begin{array}{rll} 
                &  \vdash& \langle a \rangle \langle b \rangle \psi \rightarrow \langle b \rangle \psi \\
                &  \vdash &[a](\langle a \rangle \langle b \rangle \psi \rightarrow \langle b \rangle \psi) \\
                &  \vdash &[a] \langle a \rangle \langle b \rangle \psi \rightarrow [a] \langle b \rangle \psi \\

                G_i &\vdash & [a] \langle a \rangle \langle b \rangle \psi \\

        \end{array}
        \]
        
        
        By the allowed definition we conclude $A_i \vdash \langle b \rangle \psi$ and hence $\langle b \rangle \psi \in A_i$.
