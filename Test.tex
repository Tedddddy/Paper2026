\begin{definition}
    Let $Var$ be the set of propositional variables. Then $\phi$ is defined inductively as follows: 
    \[
       \phi ::=  p \mid \phi \lor \phi \mid X\phi \mid  \phi U \psi \mid \langle a \rangle \phi 
    \]
\end{definition}

\begin{definition}
    Let  $\phi $ be a formula and $p \in Var$. Then the Closure of $\phi$ is inductively defined as

\[
    \begin{aligned}
        Cl(p) &= Cl(\neg p) = \{p,\neg p\} \\
        Cl(\neg \phi) &= Cl(\phi) \\
        Cl(\phi \lor \psi) &= \{\phi \lor \psi, \neg(\phi \lor \psi)\} \cup  Cl(\phi) \cup Cl(\psi) \\
        Cl(X \phi) &= \{X \phi, \neg X \phi\} \cup Cl(\phi) \\
        Cl(\phi U \psi) &= \{\phi U \psi, \neg(\phi U \psi)\} \cup Cl(\phi) \cup Cl(\psi) \\
        Cl(\langle a \rangle \phi) &= \{\langle a \rangle \phi, \neg \langle a \rangle \phi\} \cup Cl(\phi)
    \end{aligned}
\]
\end{definition}

\begin{definition}
An atom $A$ is a subset of $Cl(\phi)$ such that
\[
\begin{aligned}
&\text{if } \psi \in Cl(\phi),
&&\text{ then } \psi \in A \text{ iff } \neg \psi \notin A, \\
&\text{if } \phi_1 \lor \phi_2 \in Cl(\phi), 
&&\text{ then } \phi_1 \lor \phi_2 \in A\text{ iff } \phi_1 \in A \text{ or } \phi_2 \in A \\
&\text{if }  \phi_1 U \phi_2 \in Cl(\phi),
&&\text{then  if } \phi_1 U \phi_2 \in A \mbox{ then } \psi_1 \in A \mbox{ or } \psi_2 \in A \\   
&\text{if }  \phi_1 U \phi_2 \in Cl(\phi),
&&\text{then  if } \phi_2 \in A \mbox{ then } \phi_1 U \phi_2 \in A
\end{aligned}
\]
\end{definition}

\begin{definition}
    We say two atoms $A,B$ are in relation $R_x (A R_x B)$ iff

\[
\begin{aligned}
&\text{if } X\psi \in Cl(\phi),
&&\text{ then } X\psi \in A \text{ iff } \psi \in B, \\[2mm]
&\text{if } \psi_1 U \psi_2 \in Cl(\phi),
&&\text{ then if } \psi_1 U \psi_2, \neg\psi_2 \in A,
   \text{ then } \psi_1 U \psi_2 \in B, \\
&&&\text{ and if } \psi_1 U \psi_2 \in B, \psi_1 \in A,
   \text{ then } \psi_1 U \psi_2 \in A.
\end{aligned}
\]
\end{definition}

\begin{lemma}

For any atom $A$ there exists an atom $B$ s.t. $A R_x B$.

\begin{proof}
    Assume $A$ is an atom. Then we define $B'$ as follows: 
    \[
        B' = \{ \phi \mid X \phi \in A\} \cup \{ \neg \phi \mid \neg X \phi \in A\} \cup \{\psi_1 U \psi_2 \mid \psi_1 U \psi_2, \neg \psi_2 \in A\}
        \cup \{   \neg(\psi_1 U \psi_2) \mid \neg(\psi_1 U \psi_2), \psi_1 \in A  \}
    \] 

    Now we show that $B$ is consistent. Assume it is not. Then there exists $\alpha_1,..., \alpha_n$ s.t.
    $B \vdash \alpha_1 \land \alpha_2 ... \land \alpha_n \rightarrow \bot$. 
    Let's define $\hat{B} = \bigwedge B$. It follows:
    \[
        \begin{array}{rll} 
            & \vdash \hat{B} \land \alpha_1 \land ... \land\alpha_n  \rightarrow \bot \\
            & \vdash X(\hat{B} \land \alpha_1 \land ... \land\alpha_n  \rightarrow \bot )
            & \text{By X-necessitation} \\
            & \vdash X(\hat{B} \land \alpha_1 \land ... \land\alpha_n ) \rightarrow X \bot
            & \text{F2} \\
            & \vdash X(\hat{B}) \land X\alpha_1 \land ... \land X\alpha_n \rightarrow X \bot \\
        \end{array}
    \]

    Assume, $a_i = \phi$. Then by definition we have $A \vdash Xa_i$. If $a_i = \neg \phi$, then
    we can use F1 to obtain $A \vdash X \neg \phi$. If $a_i = \psi_1 U \psi_2$ then 
    by definition, $\psi_1 U \psi_2 \in A$ and $\neg \psi_2 \in A$. Consider the following proof steps:

       \[
        \begin{array}{rll} 
            & A \vdash \neg \psi_2 \\
            & A \vdash \psi_1 U \psi_2 \\
            & A \vdash \psi_1 U \psi_2 \land \neg \psi_2 \\
            & A \vdash \psi_1 U \psi_2 \land \neg \psi_2 \rightarrow (\psi_2 \lor (\psi_1 \land X ( \psi_1 U \psi_2 ))) \land \neg \psi_2 
            & \text{Unfolding U} \\
            & A \vdash  \psi_1 U \psi_2 \land \neg \psi_2 \rightarrow (\psi_1 \land X(\psi_1 U \psi_2))
            & \text{distributivity and other rules} \\
            & A \vdash \psi_1 \land X(\psi_1 U \psi_2) \\
            & A \vdash X(\psi_1 U \psi_2).

        \end{array}
        \]

        Hence, $A \vdash Xa_i$. If $a_i = \neg (\psi_1 U \psi_2)$, then by
        definition, $\neg(\psi_1 U \psi_2), \psi_1 \in A.$ Our goal is to derive
        $A \vdash \neg(\psi_1 U \psi_2) \land \psi_1 \rightarrow X \neg (\psi_1 U \psi_2)$.

        \[
        \begin{array}{rll} 
                & A \vdash \neg(\psi_1 U \psi_2) \land \psi_1 \\
                & A \vdash \neg(\psi_2 \lor (\psi_1 \land X(\psi_1 U \psi_2))) \land \psi_1
                & \text{Unfolding U} \\
                & A \vdash \neg \psi_2 \land (\neg \psi_1 \lor \neg X(\psi_1 U \psi_2)) \land \psi_1 \\
                & A \vdash \neg \psi_1 \lor \neg X(\psi_1 U \psi_2) \\
                & \neg X (\psi_1 U \psi_2) 
                & \text{$\psi_1$ is derivable} \\
                & A \vdash X \neg (\psi_1 U \psi_2) 
                & \text{F1}

        \end{array}
        \]

        
        By similar argument, we can conclude $A \vdash X\hat{B}$. So it follows 
        \[
            A \vdash X\bot
        \] 
        Because $X \bot \rightarrow \bot$ is a tautology, we conclude $A \vdash \bot$ which is a contradiction. 
        Now, by the Extension Lemma, we can extend $B'$ to an atom $B$. It is easy to check, that $A R_x B$.
        

\end{proof}

\end{lemma}

\begin{lemma}
    Assume a global path $G_0G_1...$.  Then for any $\psi \in Cl(\phi)$:

    $$  \psi \in G_i \mbox{ iff } ((G_iG_{i+1}...), (G_1...G_i)) \vDash \psi    $$

    For every agent a, every $A_0A_1...$ path in $T_a$ and for any $\psi \in Cl(\phi)$:

    $$ \psi \in A_i \mbox{ iff } ((A_iA_{i+1}...), (A_1...A_i)) \vDash \psi    $$

    \begin{proof}
        We show both on double induction on the length of $\phi$. We first prove the global case.
        Assume $\phi = p$. We define the truth on each atom to be $L(G_i) = \{p \mid p \in G_i\}$.
        Then it follows: $$p \in G_i \iff p \in L(G_i) \iff (G_i...G_\infty, G_0...G_i) \vDash p$$. If $\phi = \neg \psi$,
        then we have: $$\neg \psi \in G_i \iff \psi \notin G_i \overset{\text{IH}}{\iff} (G_i...G_\infty, G_0...G_i) \nvDash \psi \iff (G_i...G_{\infty}, G_0...G_i) \vDash \neg\psi$$.
        $\phi = \psi_1 \lor \psi_2$ is done analogously. Assume $\phi =X \psi \in G_i$. Then $X \phi \in G_i \iff \psi \in G_{i+1} \overset{\text{IH}}{\iff} 
        (G_{i+1}...G_\infty, G_0...G_{i+1}) \vDash \psi  \iff (G_i...G_\infty, G_0... G_i) \vDash X \psi$.  \newline 
        Next we deal with the Until case, so assume $\phi = \psi_1 U \psi_2$. We prove two directions.
        If $\psi_1 U \psi_2 \in G_i$, then by construction of the global path there exists some $j \geq i$ s.t. $\psi_2 \in G_j$ and for all $i \leq k < j$: $\psi_1 \in G_k$. By IH, it follows
        $(G_j...G_\infty, G_0...G_j) \vDash \psi_2$ and $(G_k...G_\infty, G_0...G_k) \vDash \psi_1$. Hence, $(G_i...G_\infty, G_0...G_i) \vDash \psi_1 U \psi_2$.
        Now assume $(G_i...G_\infty, G_0...G_i) \vDash \psi_1 U \psi_2$. By the semantics that means there is some $j \geq i$ s.t. 
        $(G_j...G_\infty, G_0...G_j) \vDash \psi_2$ and for all $i \leq k < j$: $(G_k...G_\infty, G_0...G_k) \vDash \psi_1$. Now IH gives us
        $\psi_2 \in G_j$ and hence $\psi_1 U \psi_2 \in G_j$. Because $G_{j-1} R_x G_j$, and by IH $\psi_1 \in G_{j-1}$, we conclude $\psi_1 U \psi_2 \in G_{j-1}$. Applying this argument repeatedly, we get $\psi_1 U \psi_2 \in G_i$.
        \newline Last assume $\phi = \langle a \rangle \psi$. Then by Lemma 14 there exists an infinite allowed fulfilling path $A_0A_1...$ s.t. $\phi \in A_i$. By IH, that means $(A_i...A_\infty, A_0...A_i) \vDash \psi$.
        But that also means $(G_i...G_\infty, G_0...G_i) \vDash \langle a \rangle \psi$. Conversly, if $G_i...G_\infty, G_0...G_i \vDash \langle a \rangle \psi$,
        then there exists some $A_0A_1...$ s.t. $(A_i...A_\infty, A_0...A_i) \vDash \psi$. Applying IH gives us $\psi \in A_i$. By construction of our $T_a$ $A_i$ is $G_i$-allowed.
        Assume $\neg \langle a \rangle \psi \in G_i$. Because $A_i$ is $G_i$-allowed, we have $\neg \psi \in A_i$ which is a contradiction.


        Next we proof the standpoint case. Here we only consider $\phi = \langle b \rangle \psi$ for some agent b. The other cases are similar. So let $\langle b \rangle \psi \in A_i$. Under the assumption that for all 
        $\langle b \rangle \psi \in A_i \iff \langle b \rangle G_i$ we have that $\langle b \rangle \psi \in G_i$. By Lemma 14 we can construct some $B_0B_1...$ s.t. $\psi \in B_i$. Applying IH gives us
        $(B_i...B_\infty, B_0...B_i) \vDash \psi$. Hence, $(A_i...A_\infty,A_0...A_i) \vDash \langle b \rangle \psi$.
        

    \end{proof}
\end{lemma}
